	\begin{solution}
		Hallamos la transpuesta de la matriz:
  
            \begin{math}
			      A^{T}=
			      \begin{bmatrix}
				      3 & 2   \\
				      2 & 3   \\
                        2 & -2  \\  
			      \end{bmatrix}
		\end{math}. \\
        Ahora multiplicamos la matriz A con su transpuesta: \\
            \begin{math}
			    A^{T} . A =
			      \begin{bmatrix}
				      3 & 2   \\
				      2 & 3   \\
                        2 & -2  \\  
			      \end{bmatrix} .
                    \begin{bmatrix}
				      3 & 2 & 2  \\
				      2 & 3 & -2  \\    
			      \end{bmatrix} =
                    \begin{bmatrix}
				      13 & 12 & 2  \\
				      12 & 13 & -2  \\
                        2 & -2  & 8\\  
			      \end{bmatrix}              
		  \end{math}. \\
        Ahora hallaremos el polinomio característico, buscando luego hallar los autovalores y autovectores: \\
            \begin{math}
                P_{\lambda}(A^{T}.A)= \left | A^{T}.A-\lambda I \right | \\
                P_{\lambda}(A^{T}.A)= \left |
                \begin{pmatrix}
				      13 & 12 & 2  \\
				      12 & 13 & -2  \\
                        2 & -2  & 8\\  
			    \end{pmatrix} - 
                \begin{pmatrix}
				      \lambda & 0 & 0  \\
				      0 & \lambda & 0  \\
                        0  & 0  & \lambda \\  
			    \end{pmatrix} \right | \\
                P_{\lambda}(A^{T}.A)= 
                \begin{vmatrix}
				      13 - \lambda & 12 & 2  \\
				      12 & 13 - \lambda & -2  \\
                        2 & -2  & 8 - \lambda \\  
			    \end{vmatrix} = 
                - \lambda (\lambda - 25) (\lambda - 9) \\    
            \end{math}
                Hallamos las raices igualando a cero: \\
               - \lambda (\lambda - 25) (\lambda - 9) = 0 \\
                Las raíces halladas son:\\
                \lambda _{1}=25, \lambda _{2}=9 , \lambda _{3}=0 \\
