\begin{frame}
	\begin{enumerate}\setcounter{enumi}{29}
		\item

		      Sea
		      \begin{math}
			      A=
			      \begin{bmatrix}
				      3 & 2 & 2  \\
				      2 & 3 & -2
			      \end{bmatrix}
		      \end{math}.
		      Calcule la descomposición de valores singulares de $A$.
	\end{enumerate}

	\begin{solution}
		El polinomio característico de la matriz
		\begin{math}
			A^{T}A=
			\begin{bmatrix}
				3 & 2  \\
				2 & 3  \\
				2 & -2
			\end{bmatrix}
			\begin{bmatrix}
				3 & 2 & 2  \\
				2 & 3 & -2
			\end{bmatrix}=
			\begin{bmatrix}
				13 & 12 & 2  \\
				12 & 13 & -2 \\
				2  & -2 & 8
			\end{bmatrix}
		\end{math}
		es
		\begin{equation*}
			P_{A^{T}A}
			\left(\lambda\right)=
			\begin{vmatrix}
				A^{T}A-\lambda I
			\end{vmatrix}=
			\begin{vmatrix}
				\begin{bmatrix}
					13 & 12 & 2  \\
					12 & 13 & -2 \\
					2  & -2 & 8  \\
				\end{bmatrix}-
				\lambda
				\begin{bmatrix}
					1 & 0 & 0 \\
					0 & 1 & 0 \\
					0 & 0 & 1
				\end{bmatrix}
			\end{vmatrix}=
			\begin{vmatrix}
				13 - \lambda & 12           & 2           \\
				12           & 13 - \lambda & -2          \\
				2            & -2           & 8 - \lambda \\
			\end{vmatrix}=
			-\lambda\left(\lambda-25\right)\left(\lambda-9\right).
		\end{equation*}
		Las raíces de
		\begin{math}
			P_{A^{T}A}
			\left(\lambda\right)=
			0
		\end{math}
		son los autovalores de $A^{T}A$, a saber, $\lambda_{1}=25$,
		$\lambda_{2}=9$ y $\lambda_{3}=0$.

		Hallemos una base para los autoespacios de $A^{T}A$

		\begin{equation*}
			S\left(\lambda_{i}\right)=
			\left\{
			v\in\mathbb{R}^{3}\mid
			A^{T}Av=\lambda_{i}v
			\right\}\quad\text{para $i=1,2,3$}.
		\end{equation*}
	\end{solution}

	\begin{itemize}
