\section{Pregunta N$^{\circ}$21\qquad Nombre}

\begin{frame}
	\begin{definition}[Definida positiva]
		Sea $A\in\mathbb{R}^{n\times}$ una matriz.
		Decimos que $A$ es definida positiva sii
		\begin{math}
			\forall x\in\mathbb{R}^{n}\setminus\left\{0\right\}:
			x^{T}Ax>
			0.
		\end{math}
	\end{definition}

	\begin{example}
		Sea
		\begin{math}
			M=
			\begin{bmatrix}
				2 & 0 \\
				2 & 2
			\end{bmatrix}
		\end{math}
		una matriz \alert{no simétrica}.
		Entonces, para todo
		\begin{math}
			x=
			\begin{bmatrix}
				x_{1} \\
				x_{2}
			\end{bmatrix}
			\in\mathbb{R}^{2}\setminus\left\{0\right\}
		\end{math}
		se tiene que
		\begin{equation*}
			x^{T}Mx=
			\begin{bmatrix}
				x_{1} & x_{2}
			\end{bmatrix}
			\begin{bmatrix}
				2 & 0 \\
				2 & 2
			\end{bmatrix}
			\begin{bmatrix}
				x_{1} \\
				x_{2}
			\end{bmatrix}=
			{\left(x_{1}+x_{2}\right)}^{2}+
			x^{2}_{1}+
			x^{2}_{2}>
			0.
		\end{equation*}
		$\therefore M$ es definida positiva.
	\end{example}

	\begin{theorem}
		Sea $A\in\mathbb{R}^{n\times}$ una matriz simétrica.
		Las siguientes proposiciones son equivalentes:
		\begin{itemize}
			\item

			      $A$ es definida positiva.

			\item

			      Todas las matrices esquina $A_{k}$ tienen determinante
			      positivo.

			\item

			      $A$ puede ser reducida a su forma triangular superior
			      usando solo las operaciones elementales filas del tipo un
			      múltiplo de una fila es sumada a otra fila y los
			      elementos pivote serán todos positivos.

			\item

			      $A$ tiene una factorización de Cholesky $LL^{T}$
			      (donde $L$ es triangular inferior con entradas positivas
			      en su diagonal).

			\item

			      $A$ puede factorizarse como $B^{T}B$ para alguna matriz
			      $B$ no singular.
		\end{itemize}
	\end{theorem}
\end{frame}

\begin{frame}
	\begin{enumerate}\setcounter{enumi}{20}
		\item

		      Determine si la matriz
		      \begin{math}
			      B=
			      \begin{bmatrix}
				      4 & 2 & 6  \\
				      2 & 2 & 5  \\
				      6 & 5 & 29
			      \end{bmatrix}
		      \end{math}
		      es definida positiva.
	\end{enumerate}

	\begin{solution}[Primera forma]
		Sea
		\begin{math}
			x=
			\begin{bmatrix}
				x_{1} \\
				x_{2} \\
				x_{3}
			\end{bmatrix}\in
			\mathbb{R}^{3}\setminus\left\{0\right\}
		\end{math}.
		Entonces,
		\begin{align*}
			x^{T}Bx & =
			\begin{bmatrix}
				x_{1} & x_{2} & x_{3}
			\end{bmatrix}
			\begin{bmatrix}
				4 & 2 & 6  \\
				2 & 2 & 5  \\
				6 & 5 & 29
			\end{bmatrix}
			\begin{bmatrix}
				x_{1} \\
				x_{2} \\
				x_{3}
			\end{bmatrix}      \\
			x^{T}Bx & =
			4x^{2}_{1}+
			\textcolor{DarkGreen}{2x^{2}_{2}}+
			\textcolor{DarkBlue}{29x^{2}_{3}}+
			4x_{1}x_{2}+
			12x_{1}x_{3}+
			10x_{2}x_{3}.       \\
			x^{T}Bx & =
			\textcolor{DarkOrange}{4x^{2}_{1}}+
			\textcolor{DarkGreen}{x^{2}_{2}}+
			\left\{
			\textcolor{DarkBlue}{{\left(5x_{3}\right)}^{2}}+
			2\left(x_{2}\right)\left(5x_{3}\right)+
			\textcolor{DarkGreen}{{\left(x_{2}\right)}^{2}}
			\right\}+
			4x_{1}x_{2}+
			12x_{1}x_{3}+
			\textcolor{DarkBlue}{{\left(2x_{3}\right)}^{2}}+
			\alert{9x^{2}_{1}}-
			\alert{9x^{2}_{1}}. \\
			x^{T}Bx & =
			\left\{
			{\alert{\left(3x_{1}\right)^{2}}}+
			2\left(3x_{1}\right)\left(2x_{3}\right)+
			{\left(2x_{3}\right)}^{2}
			\right\}+
			\left\{
			{\left(5x_{3}\right)}^{2}+
			2\left(x_{2}\right)\left(5x_{3}\right)+
			{\left(x_{2}\right)}^{2}
			\right\}+
			\left\{
			\textcolor{DarkOrange}{{\left(2x_{1}\right)}^{2}}+
			2\left(2x_{1}\right)\left(x_{2}\right)+
			{\left(x_{2}\right)}^{2}
			\right\}.           \\
			x^{T}Bx & =
			{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}+
			\underbrace{{\left(x_{2}+5x_{3}\right)}^{2}}_{\geq0}+
			\underbrace{{\left(2x_{1}+x_{2}\right)}^{2}}_{\geq0}.
			\label{eq:positive}\tag{$\bigstar$}
		\end{align*}
		Para verificar que~\eqref{eq:positive} es positivo, supongamos
		que~\eqref{eq:positive} es \alert{cero}, es decir,
		\begin{equation*}
			\alert{0}=
			{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}+
			\underbrace{{\left(x_{2}+5x_{3}\right)}^{2}}_{=0}+
			\underbrace{{\left(2x_{1}+x_{2}\right)}^{2}}_{=0}
		\end{equation*}
	\end{solution}
\end{frame}

\begin{frame}
	Osea,
	\begin{equation*}
		\systeme{
		2x_{1}+x_{2}=0,
		x_{2}+5x_{3}=0
		}
		\xRightarrow{\text{eliminación gaussiana}}
		\systeme{
			x_{1}=5t,
			x_{2}=-10t,
			x_{3}=2t
		},\quad
		t\in\mathbb{R}\setminus\left\{0\right\}.
	\end{equation*}
	% \begin{math}
	% 	x_{1}=\frac{-1}{2}x_{2}
	% 	x_{2}=-5x_{3}
	% \end{math}
	Pero,
	\begin{equation*}
		{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}=
			{\left(15t+4t\right)}^{2}-{\left(15t\right)}^{2}=
		\alert{136t^{2}>0}.
	\end{equation*}
	Entonces, $\forall x\in\mathbb{R}^3\setminus\left\{0\right\}$:
	\begin{equation*}
		x^{T}Bx =
			{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}+
		{\left(x_{2}+5x_{3}\right)}^{2}+
		{\left(2x_{1}+x_{2}\right)}^{2}
		\alert{>0}
	\end{equation*}

	$\therefore B$ es definida positiva.
\end{frame}
