\section{Pregunta N$^{\circ}$21\qquad Nombre}

\begin{frame}
	\begin{definition}[Definida positiva]
		Sea $A\in\mathbb{R}^{n\times}$ una matriz.
		Decimos que $A$ es definida positiva sii
		\begin{math}
			\forall x\neq0\in\mathbb{R}^{n}:
			x^{T}Ax>
			0.
		\end{math}
	\end{definition}

	\begin{block}{Matriz que no es simétrica}
		Sea
		\begin{math}
			M=
			\begin{bmatrix}
				2 & 0 \\
				2 & 2
			\end{bmatrix}
		\end{math}
		una matriz.
		Entonces, para todo
		\begin{math}
			x=
			\begin{bmatrix}
				x_{1} \\
				x_{2}
			\end{bmatrix}
		\end{math}
		no nulo se tiene que
		\begin{equation*}
			x^{T}Mx=
			\begin{bmatrix}
				x_{1} & x_{2}
			\end{bmatrix}
			\begin{bmatrix}
				2 & 0 \\
				2 & 2
			\end{bmatrix}
			\begin{bmatrix}
				x_{1} \\
				x_{2}
			\end{bmatrix}=
			{\left(x_{1}+x_{2}\right)}^{2}+
			x^{2}_{1}+
			x^{2}_{2}>
			0.
		\end{equation*}
	\end{block}

	\begin{theorem}
		Sea $A\in\mathbb{R}^{n\times}$ una matriz simétrica.
		Las siguientes proposiciones son equivalentes:
		\begin{itemize}
			\item

			      $A$ es definida positiva.

			\item

			      Todas las matrices esquina $A_{k}$ tienen determinante positivo.

			\item

			      $A$ puede ser reducida a su forma triangular superior
			      usando solo las operaciones elementales filas del tipo un
			      múltiplo de una fila es sumada a otra fila y los
			      elementos pivote serán todos positivos.
		\end{itemize}
	\end{theorem}
\end{frame}

\begin{frame}
	\begin{enumerate}\setcounter{enumi}{20}
		\item

		      Determine si la matriz
		      \begin{math}
			      B=
			      \begin{bmatrix}
				      4 & 2 & 6  \\
				      2 & 2 & 5  \\
				      6 & 5 & 29
			      \end{bmatrix}
		      \end{math}
		      es definida positiva.
	\end{enumerate}

	\begin{solution}
		Sea
		\begin{math}
			x=
			\begin{bmatrix}
				x_{1} \\
				x_{2} \\
				x_{3}
			\end{bmatrix}\in
			\mathbb{R}^{3}
		\end{math}
		no nulo.
		Entonces,
		\begin{align*}
			x^{T}Bx & =
			\begin{bmatrix}
				x_{1} & x_{2} & x_{3}
			\end{bmatrix}
			\begin{bmatrix}
				4 & 2 & 6  \\
				2 & 2 & 5  \\
				6 & 5 & 29
			\end{bmatrix}
			\begin{bmatrix}
				x_{1} \\
				x_{2} \\
				x_{3}
			\end{bmatrix}        \\
			x^{T}Bx & =
			4x^{2}_{1}+
			2x^{2}_{2}+
			29x^{2}_{3}+
			4x_{1}x_{2}+
			12x_{1}x_{3}+
			10x_{2}x_{3}.         \\
			x^{T}Bx & =
			4x^{2}_{1}+
			x^{2}_{2}+
			\left[{\left(5x_{3}\right)}^{2}+
			2\left(x_{2}\right)\left(5x_{3}\right)+
			{\left(x_{2}\right)}^{2}
			\right]+
			4x_{1}x_{2}+
			12x_{1}x_{3}+
			{\left(2x_{3}\right)}^{2}+
			9x^{2}_{1}-9x^{2}_{1} \\
			x^{T}Bx & =
			\left[
			{\left(3x_{1}\right)}^{2}+
			2\left(3x_{1}\right)\left(2x_{3}\right)+
			{\left(2x_{3}\right)}^{2}
			\right]+
			\left[{\left(5x_{3}\right)}^{2}+
			2\left(x_{2}\right)\left(5x_{3}\right)+
			{\left(x_{2}\right)}^{2}
			\right]+
			\left[
			{\left(2x_{1}\right)}^{2}+
			2\left(2x_{1}\right)\left(x_{2}\right)+
			{\left(x_{2}\right)}^{2}
			\right]               \\
			x^{T}Bx & =
			\underbrace{{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}}_{>0}+
			\underbrace{{\left(x_{2}+5x_{3}\right)}^{2}}_{\geq0}+
			\underbrace{{\left(2x_{1}+x_{2}\right)}^{2}}_{\geq0}
			\label{eq:positive}\tag{$\star$}
		\end{align*}
		Pero~\eqref{eq:positive} es positivo, porque
		\begin{math}
			{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}
		\end{math}
		es positivo para cualquier $x_{1}$ y $x_{3}$ no nulos.
	\end{solution}
\end{frame}

\begin{frame}

    \begin{math}
        {\left(x_{2}+5x_{3}\right)}^{2}  
    \end{math}
    y
    \begin{math}
        {\left(2x_{1}+x_{2}\right)}^{2}
    \end{math}
    supondremos que estos tomaran sus valores minimos.
    \begin{math}
        {\left(x_{2}+5x_{3}\right)}^{2}=0
        \\{\left(2x_{1}+x_{2}\right)}^{2}=0
    \end{math}
    \\resolviendo el sistemas nos queda
    \begin{math}
        \\x_{1}= \frac{-1}{2}x_{2}
        \\x_{2}=-5x_{3}
    \end{math}
    \\quedandonos
    \begin{math}
        \\x_{1}= 5k
        \\x_{2}=-10k
        \\x_{3}=2k
    \end{math}
    \\donde k \in R-{0}, Ahora tenemos que verificar que 
    \begin{math}
			{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}
	\end{math}
    sea positivo
    \begin{math}
			{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}=
            {\left(15k+4k\right)}^{2}-{\left(15k\right)}^{2}=
            136k^{2}
	\end{math}
    \\Verificamos que
    \begin{math}
            136k^{2}>0
	\end{math}
    , por lo tanto la expresion
    \begin{math}
			{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}>0
	\end{math}
    \\Entonces podemos asegurar que \\
    \begin{math}
        x^{T}Bx & =
			{\left(3x_{1}+2x_{3}\right)}^{2}-{\left(3x_{1}\right)}^{2}+
			{\left(x_{2}+5x_{3}\right)}^{2}+
			{\left(2x_{1}+x_{2}\right)}^{2}
			 >0  
   \end{math}
  \\ Por lo tanto, la matriz B es definida positiva
