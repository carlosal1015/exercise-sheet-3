\section{Pregunta N$^{\circ}$12\qquad Carlos Daniel Malvaceda Canales}

\begin{frame}
	\begin{enumerate}\setcounter{enumi}{11}
		\item

		      En una heladería, por $1$ helado, $2$ zumos y $4$ batidos
		      nos cobraron S/$35$.
		      Otro día, por $4$ helados, $4$ zumos y $1$ batido nos
		      cobraron S/$34$.
		      Un tercer día por $2$ helados, $3$ zumos y $4$ batidos
		      S/$42$.
		      Determine el precio de cada uno según el siguiente
		      requerimiento.

		      \begin{enumerate}[a)]
			      \item

			            Modele el sistema.

			      \item
			            Resuelve el sistema usando el método de Parlett-Reid.
		      \end{enumerate}

	\end{enumerate}

	\begin{solution}
		\begin{enumerate}[a)]
			\item

			      Sean $x$ el precio del helado, $y$ el precio del zumo y $z$ el
			      precio del batido.
			      Entonces, resulta el siguiente sistema lineal

			      \begin{equation*}
				      \systeme{
					      x+2y+4z=35,
					      4x+4y+z=34,
					      2x+3y+4z=42
				      }
			      \end{equation*}
			\item

			      En notación matricial $Ax=b$,
			      \begin{align*}
				      \underbrace{
					      \begin{bmatrix}
						      1 & 2 & 4 \\
						      4 & 4 & 1 \\
						      2 & 3 & 4
					      \end{bmatrix}
				      }_{A}
				      \underbrace{
					      \begin{bmatrix}
						      x \\
						      y \\
						      z
					      \end{bmatrix}
				      }_{x}=
				      \underbrace{
					      \begin{bmatrix}
						      35 \\
						      34 \\
						      42
					      \end{bmatrix}
				      }_{b}
			      \end{align*}.

		\end{enumerate}
		Pero \alert{$A$ no es simétrica}, por lo que no se puede aplicar
		la descomposición Parlett-Reid.
		Así que realizamos el intercambio de filas para que la matriz sea
		simétrica.
	\end{solution}
\end{frame}

\begin{frame}
	\begin{solution}
		\begin{enumerate}[b)]
			\item

			      \begin{align*}
				      \begin{bmatrix}
					      1 & 2 & 4 \\
					      2 & 3 & 4 \\
					      4 & 4 & 1
				      \end{bmatrix}
				      \begin{bmatrix}
					      x \\
					      y \\
					      z
				      \end{bmatrix} =
				      \begin{bmatrix}
					      35 \\
					      42 \\
					      34
				      \end{bmatrix}.
			      \end{align*}

			      La descomposición Parlett-Reid es de la forma
			      $PAP^{T}=LTL^{T}$, lo cual nos llevara $n-2$ pasos.

			      Ahora, $A$ es simétrica e indefinida porque tiene al menos un autovalor
			      negativo y positivo.
			      Ahora realizaremos las iteraciones:
			      \begin{description}
				      \item[Etapa $k=1$]

					      \begin{equation*}
						      A =
						      \begin{bmatrix}
							      1         & 2 & 4 \\
							      \alert{2} & 3 & 4 \\
							      \alert{4} & 4 & 1
						      \end{bmatrix}
					      \end{equation*}

					      Buscamos el elemento del vector
					      \begin{math}
						      \begin{bmatrix}
							      a_{21} & a_{31}
						      \end{bmatrix}
					      \end{math}
					      de mayor valor absoluto y se determina una permutación, que
					      representaremos por
					      \begin{math}
						      P_{1}
					      \end{math}, tal que:
					      \begin{equation*}
						      {P}_{1}
						      \begin{bmatrix}
							      a_{21} \\
							      a_{31}
						      \end{bmatrix}=
						      \begin{bmatrix}
							      \widehat{a}_{2} \\
							      \widehat{a}_{3}
						      \end{bmatrix},
					      \end{equation*}
					      donde
					      \begin{math}
						      \left|\widehat{a_{2}}\right|=
						      \max\left\{
						      \left|\widehat{a_{21}}\right|,
						      \left|\widehat{a_{31}}\right|
						      \right\}
					      \end{math}.
			      \end{description}
		\end{enumerate}
	\end{solution}
\end{frame}

\begin{frame}
	\begin{solution}
		\begin{enumerate}[b)]
			\item

			      Donde
			      \begin{math}
				      \alpha_{1}=
				      \begin{bmatrix}
					      0 \\
					      0 \\
					      \dfrac{\widehat{a_{3}}}{\widehat{a_{2}}}
				      \end{bmatrix}
			      \end{math},
			      \begin{math}
				      P_{1}=
				      \begin{bmatrix}
					      e_{1} & e_{3} & e_{2}
				      \end{bmatrix}=
				      \begin{bmatrix}
					      1 & 0 & 0 \\
					      0 & 0 & 1 \\
					      0 & 1 & 0
				      \end{bmatrix}
			      \end{math}.
			      Entonces
			      \begin{align*}
				      M_{1}              & =
				      I_{3}-\alpha_{1}e_{2}=
				      \begin{bmatrix}
					      1 & 0 & 0 \\
					      0 & 1 & 0 \\
					      0 & 0 & 1
				      \end{bmatrix}-
				      \begin{bmatrix}
					      0 \\
					      0 \\
					      \frac{2}{4}
				      \end{bmatrix}
				      \begin{bmatrix}
					      0 & 1 & 0
				      \end{bmatrix}         \\
				      A^{\left(1\right)} & =
				      M_{1}
				      P_{1}
				      A
				      P_{1}^{T}
				      M_{1}^{T}=
				      \begin{bmatrix}
					      1 & 4   & 0    \\
					      4 & 1   & 7/2  \\
					      0 & 7/2 & -3/4
				      \end{bmatrix}=T.
			      \end{align*}
			      Ya tenemos la matriz tridiagonal, se cumplieron los $n-2$ pasos.
			      Por lo tanto,
			      \begin{equation*}
				      P=P_{n-2}\cdots P_{1}=
				      \begin{bmatrix}
					      1 & 0 & 0 \\
					      0 & 0 & 1 \\
					      0 & 1 & 0
				      \end{bmatrix},\quad
				      L=
				      {\left(M_{1}P_{1}P^{T}\right)}^{-1}=
				      \begin{bmatrix}
					      1 & 0   & 0 \\
					      0 & 1   & 0 \\
					      0 & 1/2 & 1
				      \end{bmatrix}.
			      \end{equation*}
		\end{enumerate}
	\end{solution}
\end{frame}

\begin{frame}
	\begin{enumerate}[b)]
		\item

		      Mediante la factorización, resolvemos $Ax=b$ en las siguientes etapas:

		      \begin{itemize}
			      \item

			            \begin{math}
				            Lz=
				            Pb
			            \end{math}.
			      \item

			            \begin{math}
				            TW=
				            z
			            \end{math}.
			            (Las matrices tridiagonales se resuelven por el algoritmo Thomas).
			      \item

			            \begin{math}
				            L^{T}y=
				            W
			            \end{math}

			      \item

			            \begin{math}
				            x=
				            P^{T}y
			            \end{math}.
		      \end{itemize}
	\end{enumerate}

	\begin{align*}
		z=
		\begin{bmatrix}
			35 & -106 & 0.2666
		\end{bmatrix},\quad
		w=
		\begin{bmatrix}
			3 & 8 & 4
		\end{bmatrix},\quad
		y=\begin{bmatrix}
			  3 & 6 & 4
		  \end{bmatrix}.
	\end{align*}
	Estos resultados están calculados en el siguiente programa, mediante
	la resolución de matrices y sustitución para obtener $\left[x,y,z\right]$.
	Finalmente, los precios son:
	\begin{align*}
		x:\text{precio del helado} & = 3 \\
		y:\text{precio del zumo}   & = 4 \\
		z:\text{precio del batido} & = 6
	\end{align*}
\end{frame}
